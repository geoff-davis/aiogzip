root=Path('/tmp/aiogzip-quiet-calibration-20260927')

 (root/'status.json').write_text(json.dumps(record,indent=2)+'\n')
def sample():
 before=cpu_snapshot(cgroup);time.sleep(5);after=cpu_snapshot(cgroup)
 row={'time':time.time(),'foreign_cores':foreign_cores(before,after),'load':os.getloadavg()}

  record['status']='deferred';save();sys.exit(0)

for i,engine in enumerate(order):
 argv=['/usr/bin/taskset','-c','0','/tmp/aiogzip-wp5-nightly-20260927/venv/bin/python',str(frozen/'scripts/measure_b2_stream_fairness.py'),'--source-root',str(frozen),'--engine',engine,'--repeat','7','--output',str(root/f'{i}-{engine}.json')]

  try:
   while child.poll() is None:
    h=sample()

     record['status']='contaminated';save();sys.exit(0)

  minima=[r['min_seconds'] for r in rows];drift=(max(minima)/min(minima)-1)*100
  comparisons.append({'engine':engine,'case':key[0],'fixture':key[1],'direction':key[2],'minima':minima,'drift_percent':drift,'passes':drift<=3.0})
